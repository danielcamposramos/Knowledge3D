def process_task(
    self,
    task_id: str,
    test_input: Sequence[Sequence[int]],
    train_examples: List[Dict] = None,  # NEW: train examples
    top_k: int = 3
) -> TaskResult:
    """Process task using BOTH procedural baseline + TRM reasoning."""

    # 1. Get procedural candidates from baseline (train example analysis)
    procedural_candidates = []
    if train_examples:
        from knowledge3d.training.arc_agi import CandidateGenerator
        gen = CandidateGenerator(matryoshka_dim=self.router.matryoshka_dim)
        procedural_candidates = gen.generate_candidates(test_input, train_examples)

    # 2. Get TRM candidates (multi-galaxy reasoning)
    trm_candidates = self.router.route(test_input, top_k=top_k)

    # 3. Merge candidates
    all_candidates = []

    # Add procedural candidates (from train example analysis)
    for output, instruction, rpn in procedural_candidates:
        all_candidates.append({
            'output': output,
            'program': rpn,
            'type': 'procedural',
            'source': 'baseline'
        })

    # Add TRM candidates (from multi-galaxy reasoning)
    for cand in trm_candidates:
        # Compose Drawing + Grammar
        compositions = self.composer.compose(cand.drawing_program, [cand.grammar_rule])
        for prog, ptype in compositions:
            all_candidates.append({
                'output': None,  # Execute later
                'program': prog,
                'type': ptype,
                'source': 'trm'
            })

    # 4. Execute candidates that need execution
    from knowledge3d.training.arc_agi.rpn_executor import ARCRPNExecutor
    executor = ARCRPNExecutor()

    for cand in all_candidates:
        if cand['output'] is None:
            try:
                cand['output'] = executor.execute(test_input, cand['program'])
            except Exception as e:
                # Skip failed candidates
                cand['output'] = test_input  # Identity fallback

    # 5. Return best candidate (for now, just first one)
    # TODO: Add scoring/selection logic
    if all_candidates:
        best = all_candidates[0]
        result = TaskResult(
            task_id=task_id,
            best_program=best['program'],
            program_type=best['type'],
            score=0.8 if best['source'] == 'procedural' else 0.6,
            signature={'source': best['source']}
        )
        self.shadow.record(result.signature, result.best_program, result.program_type, result.score)
        self.results.append(result)
        return result

    # Fallback: empty result
    return TaskResult(task_id=task_id, best_program="", program_type="none", score=0.0, signature={})
```

def evaluate_task(pipeline: SovereignAIPipeline, task: Dict, task_id: str) -> Dict:
    test_pairs = task.get("test", [])

def _score_candidate(self, output_grid, expected_grid=None):
    """
    Score candidate quality.

    If expected_grid provided, use exact match.
    Otherwise, use heuristic scoring.
    """
    if expected_grid is not None:
        # Validation mode: exact match = 1.0, else 0.0
        return 1.0 if np.array_equal(output_grid, expected_grid) else 0.0

    # Training mode: heuristic scoring
    # Prefer non-trivial outputs (not identity transform)
    if np.array_equal(output_grid, self._input_grid):
        return 0.1  # Identity = low score

    # Prefer outputs with structure
    num_colors = len(np.unique(output_grid))
    filled_ratio = np.count_nonzero(output_grid) / output_grid.size

    score = 0.5
    if num_colors > 1:
        score += 0.2
    if 0.1 < filled_ratio < 0.9:  # Not empty, not full
        score += 0.2

    return score
```
