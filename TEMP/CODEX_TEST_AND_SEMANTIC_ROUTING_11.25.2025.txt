```python
def route(
    self,
    test_input: Sequence[Sequence[int]],
    top_k: int = 3,
    use_semantics: bool = True
) -> List[Dict]:
    """
    Route with semantic-aware candidate selection.

    Args:
        test_input: Input grid to process
        top_k: Number of candidates to return
        use_semantics: Whether to use semantic matching (default True)

    Returns:
        List of candidate programs with scores
    """
    candidates = []

    # NEW: Semantic matching (if available and enabled)
    if use_semantics and self.semantic_context is not None:
        try:
            input_array = np.array(test_input)
            matching_contexts = self.semantic_context.find_matching_contexts(
                input_array,
                top_k=top_k * 2  # Get more matches, filter later
            )

            for ctx in matching_contexts:
                candidates.append({
                    "program": ctx["program"],
                    "source": "semantic_match",
                    "score": ctx["score"],
                    "transformation_type": ctx.get("transformation_type", "unknown"),
                    "when_to_use": ctx.get("when_to_use", []),
                    "match_reason": f"Similar to: {ctx.get('input_signature', {}).get('signature_hash', 'unknown')}"
                })

            print(f"  [SEMANTIC] Found {len(matching_contexts)} context matches")
        except Exception as e:
            print(f"  [SEMANTIC] Warning: Semantic matching failed: {e}")

    # Existing: Grammar-based candidates
    # (Keep your existing grammar candidate generation here)

    # Existing: Drawing-based candidates
    # (Keep your existing drawing candidate generation here)

    # Combine and deduplicate by program content
    seen_programs = set()
    unique_candidates = []

    # Prioritize semantic matches
    for candidate in candidates:
        prog = candidate["program"]
        if prog not in seen_programs:
            seen_programs.add(prog)
            unique_candidates.append(candidate)

    # Return top-k
    return unique_candidates[:top_k]
```
