import json
from pathlib import Path

class DrawingGalaxy:
    # ... existing code ...

    def save(self, path: Path) -> None:
        """
        Save galaxy state to JSON file.

        Saves:
        - primitives
        - strokes
        - shapes
        - scenes
        - collections
        """
        state = {
            'primitives': self.primitives,
            'strokes': self.strokes,
            'shapes': self.shapes,
            'scenes': self.scenes,
            'collections': self.collections
        }

        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, 'w', encoding='utf-8') as f:
            json.dump(state, f, indent=2)

    def load(self, path: Path) -> None:
        """
        Load galaxy state from JSON file.

        If file doesn't exist, keeps bootstrap state.
        """
        if not path.exists():
            print(f"[DrawingGalaxy] No checkpoint at {path}, using bootstrap")
            return

        with open(path, 'r', encoding='utf-8') as f:
            state = json.load(f)

        self.primitives = state.get('primitives', self.primitives)
        self.strokes = state.get('strokes', self.strokes)
        self.shapes = state.get('shapes', self.shapes)
        self.scenes = state.get('scenes', self.scenes)
        self.collections = state.get('collections', self.collections)

        print(f"[DrawingGalaxy] Loaded {len(self.shapes)} shapes from {path}")

    def add_discovered_shape(self, shape: Dict) -> None:
        """Add discovered shape (called by DualShadowCopy)."""
        if shape not in self.shapes:
            self.shapes.append(shape)
```

import json
from pathlib import Path

class GrammarGalaxy:
    # ... existing code ...

    def save(self, path: Path) -> None:
        """
        Save galaxy state to JSON file.

        Saves all grammar rules (bootstrap + discovered).
        """
        rules_data = []
        for rule in self.rules:
            rules_data.append({
                'rule_id': rule.rule_id,
                'rpn_program': rule.rpn_program,
                'pattern': rule.pattern,
                'domain': getattr(rule, 'domain', 'general'),
                'examples': rule.examples
            })

        state = {
            'rules': rules_data,
            'total_count': len(self.rules)
        }

        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, 'w', encoding='utf-8') as f:
            json.dump(state, f, indent=2)

    def load(self, path: Path) -> None:
        """
        Load galaxy state from JSON file.

        If file doesn't exist, keeps bootstrap 196 rules.
        """
        if not path.exists():
            print(f"[GrammarGalaxy] No checkpoint at {path}, using bootstrap (196 rules)")
            return

        with open(path, 'r', encoding='utf-8') as f:
            state = json.load(f)

        # Replace rules with loaded state
        from knowledge3d.training.arc_agi.grammar_galaxy import GrammarRule

        self.rules = []
        for rule_data in state.get('rules', []):
            rule = GrammarRule(
                rule_id=rule_data['rule_id'],
                rpn_program=rule_data['rpn_program'],
                pattern=rule_data['pattern'],
                examples=rule_data['examples']
            )
            # Restore domain if present
            if 'domain' in rule_data:
                rule.domain = rule_data['domain']
            self.rules.append(rule)

        print(f"[GrammarGalaxy] Loaded {len(self.rules)} rules from {path}")
```

### Task 3: Add Persistence to DualShadowCopy (30 min)

**File**: `knowledge3d/training/arc_agi/dual_shadow_copy.py`

**Add methods**:
```python
import json
from pathlib import Path

class DualShadowCopy:
    # ... existing code ...

    def save(self, path: Path) -> None:
        """
        Save shadow copy library to JSON file.

        Saves all successful patterns recorded.
        """
        state = {
            'library': self.library,
            'pending': self.pending if hasattr(self, 'pending') else [],
            'total_entries': len(self.library)
        }

        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, 'w', encoding='utf-8') as f:
            json.dump(state, f, indent=2)

    def load(self, path: Path) -> None:
        """
        Load shadow copy library from JSON file.

        If file doesn't exist, starts with empty library.
        """
        if not path.exists():
            print(f"[DualShadowCopy] No checkpoint at {path}, starting fresh")
            return

        with open(path, 'r', encoding='utf-8') as f:
            state = json.load(f)

        self.library = state.get('library', [])
        if hasattr(self, 'pending'):
            self.pending = state.get('pending', [])

        print(f"[DualShadowCopy] Loaded {len(self.library)} shadow entries from {path}")
```

    pipeline.drawing.save(DRAWING_CHECKPOINT)
    pipeline.grammar.save(GRAMMAR_CHECKPOINT)
    pipeline.shadow.save(SHADOW_CHECKPOINT)
