3. Update `record()` signature:
```python
def record(
    self,
    signature,
    program,
    program_type,
    score,
    input_grid=None,   # NEW
    output_grid=None,  # NEW
    task_id=None       # NEW
):
    # 1. Quality scoring
    quality_score = self.quality_scorer.score_quality(program, score)

    # 2. Quality threshold
    if quality_score < 0.65:
        return

    # 3. Semantic context (if grids provided)
    semantic_ctx = None
    if input_grid is not None and output_grid is not None:
        semantic_ctx = self.semantic_context.record_context(
            program, np.array(input_grid), np.array(output_grid),
            task_id or "unknown", quality_score
        )

    # 4. Deduplication
    prog_hash, is_new = self.deduplicator.add_or_reference(
        program, program_type, quality_score,
        context=semantic_ctx
    )

    # 5. Record if new
    if is_new:
        complexity = self.quality_scorer.get_complexity_level(program)
        print(f"  [NEW] {complexity}: {program[:60]} (Q={quality_score:.2f})")

        entry = {
            "hash": prog_hash,
            "program": program,
            "type": program_type,
            "quality_score": quality_score,
            "complexity": complexity
        }

        if semantic_ctx:
            entry["semantic_context"] = {
                "transformation_type": semantic_ctx["transformation_type"],
                "when_to_use": semantic_ctx["when_to_use"],
                "signature_hash": semantic_ctx["input_signature"]["signature_hash"]
            }

        if self.staged:
            self.pending.append(entry)
        else:
            self.library.append(entry)
    else:
        stats = self.deduplicator.get_usage_stats(prog_hash)
        print(f"  [DUP] {stats['usage_count']} uses (avg={stats['avg_score']:.2f})")
```

4. Add pruning:
```python
def prune_low_quality(self) -> Dict[str, int]:
    removed = self.deduplicator.prune_low_quality(min_usage=2, min_score=0.65)
    valid_hashes = set(self.deduplicator.canonical_programs.keys())
    original = len(self.library)
    self.library = [e for e in self.library if e.get("hash") in valid_hashes]
    return {
        "removed_programs": removed,
        "removed_from_library": original - len(self.library),
        "unique_remaining": len(self.deduplicator.canonical_programs)
    }
```
