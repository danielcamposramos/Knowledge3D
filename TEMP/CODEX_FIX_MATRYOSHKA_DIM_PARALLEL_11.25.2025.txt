```python
class MatryoshkaTRM:
    """Matryoshka Temporal Reasoning Machine."""

    def __init__(
        self,
        dims: Optional[List[int]] = None,
        device: str = "cuda"
    ):
        # Use provided dimensions
        if dims is None:
            dims = [64, 128, 256]  # ← Conservative default (not 2048!)

        self.dims = dims
        self.device = device

        # Allocate ONLY for requested dimensions
        self.embedders = {}
        for dim in dims:
            self.embedders[dim] = self._create_embedder(dim)

        total_params = sum(
            sum(p.numel() for p in emb.parameters())
            for emb in self.embedders.values()
        )

        print(f"[MatryoshkaTRM] Initialized with dims={dims}")
        print(f"[MatryoshkaTRM] Total parameters: {total_params:,} "
              f"(~{total_params * 4 / 1e9:.2f} GB)")
```

```python
def run_epoch(pipeline, tasks, args):
    """Run one epoch with GPU parallelization."""

    results = []

    # OPTION 1: Batch processing (simpler)
    # Process tasks in batches for better GPU utilization
    batch_size = 3  # Process 3 tasks concurrently

    for batch_start in range(0, len(tasks), batch_size):
        batch_tasks = tasks[batch_start:batch_start + batch_size]

        # Process batch (GPU will parallelize internally)
        for task in batch_tasks:
            result = pipeline.process_task(
                task_id=task['task_id'],
                test_input=task['test'][0]['input'],
                train_examples=task['train'],
                expected_output=task['test'][0].get('output'),
                top_k=args.top_k
            )
            results.append(result)

    return results
```

```python
import torch.multiprocessing as mp
from concurrent.futures import ThreadPoolExecutor

def run_epoch_parallel(pipeline, tasks, args):
    """Run epoch with parallel GPU execution."""

    # Use ThreadPoolExecutor for I/O-bound operations
    # GPU ops will parallelize automatically via CUDA streams

    def process_single_task(task):
        try:
            return pipeline.process_task(
                task_id=task['task_id'],
                test_input=task['test'][0]['input'],
                train_examples=task['train'],
                expected_output=task['test'][0].get('output'),
                top_k=args.top_k
            )
        except Exception as e:
            print(f"[ERROR] Task {task['task_id'][:8]} failed: {e}")
            return None

    # Parallel execution (3-4 workers for 12GB GPU)
    with ThreadPoolExecutor(max_workers=3) as executor:
        results = list(executor.map(process_single_task, tasks))

    return [r for r in results if r is not None]
```

```python
import torch

class AdaptiveSwarmTRM:
    """Adaptive Swarm TRM with CUDA stream parallelization."""

    def __init__(self, max_dims, num_workers=3, device="cuda"):
        self.max_dims = max_dims
        self.num_workers = num_workers
        self.device = device

        # Create CUDA streams for parallel execution
        if device == "cuda" and torch.cuda.is_available():
            self.streams = [torch.cuda.Stream() for _ in range(num_workers)]
        else:
            self.streams = [None] * num_workers

        # ... rest of init ...

    def process_batch(self, inputs):
        """Process batch of inputs in parallel across CUDA streams."""

        if not self.streams[0]:
            # No CUDA, sequential fallback
            return [self._process_single(inp) for inp in inputs]

        # Parallel processing across CUDA streams
        results = []

        for i, inp in enumerate(inputs):
            stream_idx = i % self.num_workers
            stream = self.streams[stream_idx]

            with torch.cuda.stream(stream):
                result = self._process_single(inp)
                results.append(result)

        # Synchronize all streams
        for stream in self.streams:
            stream.synchronize()

        return results
```
