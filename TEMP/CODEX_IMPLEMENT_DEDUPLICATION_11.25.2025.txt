**Implementation**:

#### File: `knowledge3d/training/arc_agi/content_deduplicator.py` (NEW)

```python
"""
Content-based deduplication for RPN programs.
Uses SHA256 hashing for canonical references.
"""

import hashlib
import json
from pathlib import Path
from typing import Dict, List, Optional


class ContentDeduplicator:
    """
    Deduplicate RPN programs using content hashing.

    Stores:
    1. Canonical programs (unique content)
    2. Reference index (hash → canonical ID)
    3. Usage metadata (when/where each program worked)
    """

    def __init__(self):
        self.canonical_programs: Dict[str, Dict] = {}  # hash → program data
        self.reference_index: Dict[str, str] = {}      # alias → canonical hash
        self.usage_metadata: Dict[str, List[Dict]] = {}  # hash → usage records

    def compute_hash(self, program: str) -> str:
        """Compute SHA256 hash of normalized RPN program."""
        # Normalize whitespace
        normalized = " ".join(program.split())
        return hashlib.sha256(normalized.encode('utf-8')).hexdigest()[:16]

    def add_or_reference(
        self,
        program: str,
        program_type: str,
        score: float,
        context: Optional[Dict] = None
    ) -> tuple[str, bool]:
        """
        Add program or return reference to existing.

        Returns:
            (canonical_hash, is_new)
        """
        prog_hash = self.compute_hash(program)

        # Check if already exists
        if prog_hash in self.canonical_programs:
            # Update usage metadata
            usage_record = {
                "score": score,
                "context": context or {}
            }
            self.usage_metadata[prog_hash].append(usage_record)
            return prog_hash, False  # Not new

        # Add as new canonical program
        self.canonical_programs[prog_hash] = {
            "hash": prog_hash,
            "program": program,
            "type": program_type,
            "first_seen_score": score,
            "usage_count": 1
        }

        self.usage_metadata[prog_hash] = [{
            "score": score,
            "context": context or {}
        }]

        return prog_hash, True  # New program

    def get_program(self, prog_hash: str) -> Optional[Dict]:
        """Retrieve canonical program by hash."""
        return self.canonical_programs.get(prog_hash)

    def get_usage_stats(self, prog_hash: str) -> Dict:
        """Get usage statistics for a program."""
        usage_records = self.usage_metadata.get(prog_hash, [])

        if not usage_records:
            return {}

        scores = [r["score"] for r in usage_records]
        return {
            "usage_count": len(usage_records),
            "avg_score": sum(scores) / len(scores),
            "max_score": max(scores),
            "min_score": min(scores)
        }

    def prune_low_quality(self, min_usage: int = 2, min_score: float = 0.6) -> int:
        """
        Remove programs with low usage/quality.

        Returns:
            Number of programs removed
        """
        to_remove = []

        for prog_hash, program in self.canonical_programs.items():
            stats = self.get_usage_stats(prog_hash)

            # Remove if:
            # 1. Used only once AND score < 0.8 (likely noise)
            # 2. Average score < min_score across all uses
            if (stats["usage_count"] < min_usage and stats["max_score"] < 0.8) or \
               (stats["avg_score"] < min_score):
                to_remove.append(prog_hash)

        for prog_hash in to_remove:
            del self.canonical_programs[prog_hash]
            del self.usage_metadata[prog_hash]

        return len(to_remove)

    def save(self, path: Path) -> None:
        """Save deduplication index to JSON."""
        state = {
            "canonical_programs": self.canonical_programs,
            "usage_metadata": self.usage_metadata,
            "total_unique": len(self.canonical_programs),
            "total_references": sum(len(u) for u in self.usage_metadata.values())
        }

        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, 'w', encoding='utf-8') as f:
            json.dump(state, f, indent=2)

    def load(self, path: Path) -> None:
        """Load deduplication index from JSON."""
        if not path.exists():
            print(f"[ContentDeduplicator] No checkpoint at {path}, starting fresh")
            return

        with open(path, 'r', encoding='utf-8') as f:
            state = json.load(f)

        self.canonical_programs = state.get("canonical_programs", {})
        self.usage_metadata = state.get("usage_metadata", {})

        print(f"[ContentDeduplicator] Loaded {len(self.canonical_programs)} unique programs "
              f"({state.get('total_references', 0)} total references)")
```

### Layer 2: Quality-Based Filtering

**Goal**: Prioritize compound, novel transformations over simple duplicates

#### File: `knowledge3d/training/arc_agi/quality_scorer.py` (NEW)

```python
"""
Quality scoring for discovered RPN programs.
Rewards compound transformations, novelty, semantic diversity.
"""

import re
from typing import Dict


class QualityScorer:
    """
    Score program quality based on:
    1. Complexity (compound vs simple)
    2. Novelty (new pattern vs seen before)
    3. Semantic diversity (different visual effect)
    """

    SIMPLE_PATTERNS = {
        r'^\d+ rotate$',           # Just rotation
        r'^\d+ flip$',             # Just flip
        r'^\d+ \d+ RECOLOR$',      # Just recolor
        r'^IDENTITY$',             # Identity
    }

    COMPOUND_INDICATORS = [
        'compose', 'COMPOSE',
        'then', 'THEN',
        'and', 'AND',
    ]

    def __init__(self):
        self.seen_patterns = set()  # Track unique structural patterns

    def score_quality(self, program: str, execution_score: float) -> float:
        """
        Compute quality score (0.0-1.0).

        Combines:
        - Execution score (does it work?)
        - Complexity bonus (is it compound?)
        - Novelty bonus (is it new?)
        """
        # Base score from execution
        quality = execution_score

        # Penalty for simple patterns
        is_simple = any(re.match(pattern, program) for pattern in self.SIMPLE_PATTERNS)
        if is_simple:
            quality *= 0.5  # 50% penalty

        # Bonus for compound transformations
        is_compound = any(indicator in program for indicator in self.COMPOUND_INDICATORS)
        if is_compound:
            quality = min(quality * 1.3, 1.0)  # 30% bonus (capped)

        # Bonus for novel structural patterns
        pattern_signature = self._extract_pattern_signature(program)
        if pattern_signature not in self.seen_patterns:
            quality = min(quality * 1.2, 1.0)  # 20% novelty bonus
            self.seen_patterns.add(pattern_signature)

        return quality

    def _extract_pattern_signature(self, program: str) -> str:
        """
        Extract structural signature (ignoring specific values).

        Example:
            "1 rotate" → "N rotate"
            "2 6 RECOLOR" → "N N RECOLOR"
            "compose 1 rotate 2 flip" → "compose N rotate N flip"
        """
        tokens = program.split()
        signature = []

        for token in tokens:
            if token.isdigit():
                signature.append('N')  # Normalize numbers
            else:
                signature.append(token)

        return ' '.join(signature)

    def get_complexity_level(self, program: str) -> str:
        """Classify program complexity."""
        is_simple = any(re.match(pattern, program) for pattern in self.SIMPLE_PATTERNS)
        is_compound = any(indicator in program for indicator in self.COMPOUND_INDICATORS)

        if is_compound:
            return "compound"
        elif is_simple:
            return "simple"
        else:
            return "intermediate"
```

```python
def record(self, signature, program, program_type, score, context=None):
    """
    Record discovery with deduplication + quality filtering.

    Process:
    1. Compute quality score (complexity + novelty)
    2. Check if program is duplicate (content hash)
    3. If new, add to canonical library
    4. If duplicate, update usage metadata
    5. Only persist if quality ≥ threshold
    """
    # Step 1: Compute quality score
    quality_score = self.quality_scorer.score_quality(program, score)

    # Step 2: Quality threshold (stricter than execution threshold)
    if quality_score < 0.65:  # Slightly higher than 0.6 execution threshold
        return  # Skip low-quality discoveries

    # Step 3: Deduplication check
    prog_hash, is_new = self.deduplicator.add_or_reference(
        program=program,
        program_type=program_type,
        score=quality_score,
        context=context or {}
    )

    # Step 4: Only log new discoveries
    if is_new:
        complexity = self.quality_scorer.get_complexity_level(program)
        print(f"  [NEW] {complexity} program: {program[:80]} (quality={quality_score:.2f})")

        # Add to library (canonical reference)
        entry = {
            "hash": prog_hash,
            "program": program,
            "type": program_type,
            "quality_score": quality_score,
            "complexity": complexity,
            "signature": signature
        }

        if self.staged:
            self.pending.append(entry)
        else:
            self.library.append(entry)
    else:
        # Duplicate - just update metadata
        stats = self.deduplicator.get_usage_stats(prog_hash)
        print(f"  [DUP] Program seen {stats['usage_count']} times (avg_score={stats['avg_score']:.2f})")
```

```python
def prune_low_quality(self) -> Dict[str, int]:
    """
    Remove low-quality duplicates.

    Returns:
        Stats on pruned items
    """
    # Prune from deduplicator
    removed_programs = self.deduplicator.prune_low_quality(
        min_usage=2,
        min_score=0.65
    )

    # Remove from library
    valid_hashes = set(self.deduplicator.canonical_programs.keys())
    original_size = len(self.library)
    self.library = [e for e in self.library if e.get("hash") in valid_hashes]
    removed_from_library = original_size - len(self.library)

    return {
        "removed_programs": removed_programs,
        "removed_from_library": removed_from_library,
        "unique_remaining": len(self.deduplicator.canonical_programs),
        "library_size": len(self.library)
    }
```
