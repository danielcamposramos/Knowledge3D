**Execute matched grammar rules to generate candidates**

```python
class ProceduralCandidateGenerator:
    """Generate candidates by EXECUTING grammar rule RPN programs."""

    def __init__(self):
        self.rpn_engine = ModularRPNEngine()
        self.semantic_compiler = SemanticCompiler()
        self.rpn_executor = RPNExecutor()

    def generate_candidates(self, test_input, matched_rules):
        """
        Generate candidates by executing grammar rule RPN programs.

        NO TRAINING! Just EXECUTE the procedural rules!
        """
        candidates = []

        for rule in matched_rules:
            # Parse rule's RPN program
            semantic = self.semantic_compiler.compile(rule.rpn_program, test_input)

            # Execute RPN program (sovereign GPU!)
            output = self.rpn_executor.execute(semantic, test_input)

            candidates.append((output, rule.rule_id, rule.rpn_program))

        return candidates
```

### 5. Shadow Copy (STORE SUCCESSFUL RPN!)
**Store successful (task → grammar_rule) mappings**

```python
class ShadowCopyLearner:
    """Store successful task → grammar_rule mappings (PROCEDURAL!)."""

    def __init__(self, grammar_galaxy: GrammarGalaxy):
        self.grammar = grammar_galaxy
        self.success_library = []  # (task_sig, rule_id, score)

    def record_success(self, task_signature, rule_id, score):
        """Record successful task → rule mapping."""
        self.success_library.append({
            "task_signature": task_signature,
            "rule_id": rule_id,
            "score": score
        })

    def query_similar_tasks(self, task_signature, k=5):
        """
        Find similar successful tasks (k-NN lookup).

        NO ML! Just procedural similarity matching!
        """
        similarities = []

        for entry in self.success_library:
            sim = self._compute_similarity(task_signature, entry["task_signature"])
            similarities.append((entry["rule_id"], sim))

        similarities.sort(key=lambda x: x[1], reverse=True)

        return [rule_id for rule_id, sim in similarities[:k]]

    def _compute_similarity(self, sig1, sig2):
        """Compute similarity between task signatures (procedural!)."""
        score = 0.0
        if sig1.get("grid_shape") == sig2.get("grid_shape"):
            score += 0.4
        if sig1.get("num_colors") == sig2.get("num_colors"):
            score += 0.3
        if sig1.get("pattern_type") == sig2.get("pattern_type"):
            score += 0.3
        return score
```
