**Enhanced matching** (multi-tier thresholds):
```python
def find_matching_contexts(
    self,
    query_grid,
    top_k=5,
    similarity_threshold=0.5,  # ← Lowered from 0.7
    use_fuzzy_matching=True
):
    """Find matching contexts with multi-tier similarity."""
    query_sig = SemanticSignature.extract(query_grid)

    matches = []
    for ctx in self.contexts:
        input_sig = ctx["input_signature"]

        # Multi-component similarity
        structural_sim = self._structural_similarity(query_sig, input_sig)
        color_sim = self._color_similarity(query_sig, input_sig)
        pattern_sim = self._pattern_similarity(query_sig, input_sig)

        # Weighted combination
        similarity = (
            0.4 * structural_sim +  # Shape/symmetry
            0.3 * pattern_sim +      # Connected components
            0.3 * color_sim          # Color distribution
        )

        # Fuzzy matching (optional)
        if use_fuzzy_matching and similarity < similarity_threshold:
            # Try relaxed matching
            if self._fuzzy_match(query_sig, input_sig):
                similarity = max(similarity, 0.45)  # Boost fuzzy matches

        if similarity >= similarity_threshold:
            matches.append({
                "program": ctx["program"],
                "score": similarity,
                "transformation_type": ctx.get("transformation_type"),
                "when_to_use": ctx.get("when_to_use", []),
                "match_components": {
                    "structural": structural_sim,
                    "color": color_sim,
                    "pattern": pattern_sim
                }
            })

    # Sort by score and return top-k
    matches.sort(key=lambda x: x["score"], reverse=True)
    return matches[:top_k]

def _structural_similarity(self, sig1, sig2) -> float:
    """Compare structural features (shape, symmetry, sparsity)."""
    score = 0.0

    # Symmetry matching
    for sym_type in ["symmetry_vertical", "symmetry_horizontal", "symmetry_diagonal"]:
        if sig1.get(sym_type) == sig2.get(sym_type):
            score += 0.33

    # Sparsity similarity (within 20%)
    sparsity_diff = abs(sig1.get("sparsity", 0.5) - sig2.get("sparsity", 0.5))
    if sparsity_diff < 0.2:
        score += 0.2

    return min(score, 1.0)

def _color_similarity(self, sig1, sig2) -> float:
    """Compare color features (num_colors, distribution)."""
    score = 0.0

    # Number of colors (exact match)
    if sig1.get("num_colors") == sig2.get("num_colors"):
        score += 0.5

    # Color distribution similarity (Jaccard)
    colors1 = set(sig1.get("color_distribution", {}).keys())
    colors2 = set(sig2.get("color_distribution", {}).keys())

    if colors1 and colors2:
        jaccard = len(colors1 & colors2) / len(colors1 | colors2)
        score += 0.5 * jaccard

    return score

def _pattern_similarity(self, sig1, sig2) -> float:
    """Compare pattern features (components, borders, repetition)."""
    score = 0.0

    # Connected components (within ±2)
    comp_diff = abs(sig1.get("connected_components", 0) - sig2.get("connected_components", 0))
    if comp_diff <= 2:
        score += 0.4

    # Border presence
    if sig1.get("has_border") == sig2.get("has_border"):
        score += 0.3

    # Repetition pattern
    if sig1.get("has_repetition") == sig2.get("has_repetition"):
        score += 0.3

    return min(score, 1.0)

def _fuzzy_match(self, sig1, sig2) -> bool:
    """Fuzzy matching for edge cases."""
    # Same transformation type likely applicable
    if sig1.get("transformation_type") == sig2.get("transformation_type"):
        return True

    # Similar dimensions (within 50%)
    dims1 = sig1.get("dimensions", (0, 0))
    dims2 = sig2.get("dimensions", (0, 0))

    if dims1[0] > 0 and dims2[0] > 0:
        ratio = min(dims1[0] / dims2[0], dims2[0] / dims1[0])
        if ratio > 0.5:
            return True

    return False
```

**Why this works**:
- Multi-component similarity (structural, color, pattern) → more robust
- Lower threshold (0.5 vs 0.7) → more matches
- Fuzzy matching fallback → handle edge cases
- Component scores visible → debuggable

```python
@staticmethod
def extract(grid: np.ndarray, multi_scale=True) -> Dict:
    """Extract multi-scale semantic signature from grid."""
    h, w = grid.shape

    # Basic features (existing)
    signature = {
        "dimensions": (h, w),
        "num_colors": len(np.unique(grid)),
        "sparsity": float(np.sum(grid == 0) / grid.size),
        "symmetry_vertical": SemanticSignature._check_symmetry_vertical(grid),
        "symmetry_horizontal": SemanticSignature._check_symmetry_horizontal(grid),
        "symmetry_diagonal": SemanticSignature._check_symmetry_diagonal(grid),
    }

    # Color features (enhanced)
    colors, counts = np.unique(grid, return_counts=True)
    signature["color_distribution"] = {
        int(color): float(count / grid.size)
        for color, count in zip(colors, counts)
    }
    signature["background_color"] = int(colors[np.argmax(counts)])
    signature["foreground_colors"] = [int(c) for c in colors if c != signature["background_color"]]

    # Pattern features (enhanced)
    signature["connected_components"] = SemanticSignature._count_connected_components(grid)
    signature["has_border"] = SemanticSignature._has_border(grid)
    signature["has_repetition"] = SemanticSignature._detect_repetition(grid)

    # NEW: Multi-scale features
    if multi_scale and h > 3 and w > 3:
        signature["scale_2x2"] = SemanticSignature._downsample_signature(grid, 2)
        signature["scale_4x4"] = SemanticSignature._downsample_signature(grid, 4)

    # NEW: Topological invariants
    signature["euler_characteristic"] = SemanticSignature._compute_euler_char(grid)
    signature["betti_numbers"] = SemanticSignature._compute_betti_numbers(grid)

    # NEW: Transformation type hints
    signature["transformation_type"] = SemanticSignature._infer_transformation_type(signature)

    # Signature hash (for fast lookup)
    signature["signature_hash"] = SemanticSignature._compute_hash(signature)

    return signature

@staticmethod
def _downsample_signature(grid: np.ndarray, factor: int) -> Dict:
    """Extract signature from downsampled grid (coarse features)."""
    h, w = grid.shape
    if h < factor or w < factor:
        return {}

    # Simple downsampling (take every Nth pixel)
    downsampled = grid[::factor, ::factor]

    return {
        "num_colors": len(np.unique(downsampled)),
        "sparsity": float(np.sum(downsampled == 0) / downsampled.size)
    }

@staticmethod
def _compute_euler_char(grid: np.ndarray) -> int:
    """Compute Euler characteristic (topological invariant)."""
    # Simplified: V - E + F (vertices - edges + faces)
    # For binary grid: count regions - holes
    foreground = grid > 0
    components = SemanticSignature._count_connected_components(foreground)
    holes = SemanticSignature._count_holes(foreground)
    return components - holes

@staticmethod
def _count_holes(binary_grid: np.ndarray) -> int:
    """Count holes in binary grid."""
    # Find connected components of background
    inverted = ~binary_grid
    bg_components = SemanticSignature._count_connected_components(inverted)
    # Holes = interior background components (not touching border)
    # Simplified: bg_components - 1 (exterior)
    return max(0, bg_components - 1)

@staticmethod
def _compute_betti_numbers(grid: np.ndarray) -> Dict:
    """Compute Betti numbers (topological features)."""
    foreground = grid > 0
    return {
        "beta_0": SemanticSignature._count_connected_components(foreground),  # Components
        "beta_1": SemanticSignature._count_holes(foreground)  # Holes
    }

@staticmethod
def _infer_transformation_type(signature: Dict) -> str:
    """Infer likely transformation type from signature features."""
    # Rotation/reflection: asymmetric → symmetric or vice versa
    has_symmetry = any([
        signature.get("symmetry_vertical"),
        signature.get("symmetry_horizontal"),
        signature.get("symmetry_diagonal")
    ])

    if not has_symmetry:
        return "rotation_or_reflection"

    # Color change: multiple colors
    if signature.get("num_colors", 0) > 2:
        return "color_transformation"

    # Pattern repetition
    if signature.get("has_repetition"):
        return "pattern_repetition"

    # Default
    return "geometric_transformation"
```

**Why this works**:
- Multi-scale features → capture patterns at different resolutions
- Topological invariants (Euler char, Betti numbers) → robust to deformations
- Transformation type hints → better routing
- More features → better discrimination and generalization
