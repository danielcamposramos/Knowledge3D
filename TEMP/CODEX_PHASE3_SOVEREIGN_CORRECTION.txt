"""
Router Specialist: Task embedding → Transformation family.

Uses K3D's sovereign MatryoshkaTRM + SelfUpdatingAdapter.
NO PyTorch, NO external frameworks - pure PTX + RPN!
"""

import numpy as np
from pathlib import Path
from typing import Dict, List, Tuple, Optional

from knowledge3d.cranium.matryoshka_trm import MatryoshkaTRM
from knowledge3d.cranium.trm_adapters import SelfUpdatingAdapter, AdapterConfig

class RouterSpecialist:
    """
    Router Specialist for ARC-AGI.

    Architecture:
    - Base: MatryoshkaTRM (512D Matryoshka embeddings)
    - Adapter: SelfUpdatingAdapter (rank 64, sovereign!)
    - Output: 12 transformation families

    Sovereign: Uses PTX kernels via RPNMathCore, NOT PyTorch!
    """

    def __init__(self, base_trm: Optional[MatryoshkaTRM] = None, rank: int = 64):
        # Base TRM (shared, Matryoshka embeddings)
        if base_trm is None:
            base_trm = MatryoshkaTRM(max_dims=512, min_dims=64)
        self.base = base_trm

        # Transformation families (12 total)
        self.families = [
            "rotation", "flip", "translation", "recolor",
            "fill", "extract", "math", "symmetry",
            "repeat", "scale", "draw", "compose"
        ]

        # Router adapter (sovereign, rank 64)
        adapter_config = AdapterConfig(
            rank=rank,
            alpha=1.0,
            learning_rate=0.001,
            require_gpu=True  # Enforce sovereign GPU path!
        )

        self.router_adapter = SelfUpdatingAdapter(
            shape=(512, len(self.families)),  # 512D → 12 families
            rank=rank,
            specialist_name='arc_router',
            config=adapter_config
        )

        # Shadow copy: Successful routing patterns
        self.success_patterns = []  # (task_signature, family, confidence)

    def route(self, task_embedding: np.ndarray, top_k: int = 2) -> List[Tuple[str, float]]:
        """
        Route task to top-k transformation families.

        Args:
            task_embedding: (512,) numpy array
            top_k: Return top-k families

        Returns:
            families: List of (family_name, confidence) tuples
        """
        # Base TRM encodes task (Matryoshka 512D)
        # (In sovereign path, this would use GPU via bridge)
        base_features = task_embedding  # Simplified for now

        # Router adapter predicts families (sovereign GPU path!)
        logits = self.router_adapter.forward(base_features.reshape(1, -1))

        # Softmax (can use RPN for this too!)
        exp_logits = np.exp(logits - np.max(logits))
        probs = exp_logits / np.sum(exp_logits)

        # Top-k families
        top_k_indices = np.argsort(probs[0])[-top_k:][::-1]

        families = [
            (self.families[idx], float(probs[0, idx]))
            for idx in top_k_indices
        ]

        return families

    def record_success(self, task_signature: Dict, family_idx: int, confidence: float):
        """Record successful routing (shadow copy)."""
        self.success_patterns.append({
            "signature": task_signature,
            "family": family_idx,
            "family_name": self.families[family_idx],
            "confidence": confidence
        })

    def save(self, path: str):
        """Save router adapter + shadow copy."""
        import json

        # Save adapter weights (sovereign format!)
        adapter_path = Path(path).with_suffix('.adapter.npz')
        self.router_adapter.save_to_disk(str(adapter_path))

        # Save success patterns
        patterns_path = Path(path).with_suffix('.patterns.json')
        with open(patterns_path, 'w') as f:
            json.dump({
                "families": self.families,
                "success_patterns": self.success_patterns
            }, f, indent=2)

    def load(self, path: str):
        """Load router adapter + shadow copy."""
        import json

        # Load adapter weights
        adapter_path = Path(path).with_suffix('.adapter.npz')
        self.router_adapter.load_from_disk(str(adapter_path))

        # Load success patterns
        patterns_path = Path(path).with_suffix('.patterns.json')
        with open(patterns_path, 'r') as f:
            data = json.load(f)
            self.success_patterns = data["success_patterns"]
