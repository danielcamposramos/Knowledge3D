"""
Semantic context using procedural character composition.
Words are character sequences (references), not duplicate strings!
"""

import json
from pathlib import Path
from typing import Dict, List, Optional
import numpy as np

from knowledge3d.training.arc_agi.semantic_signature import SemanticSignature


class SemanticWord:
    """
    Semantic word as character sequence (procedural composition).

    Instead of storing "rotation_task" as string 250 times,
    store once as character sequence, reference 250 times.
    """

    def __init__(self, word_id: str, characters: List[int], meaning: str, category: str):
        self.word_id = word_id
        self.characters = characters  # Character IDs (references to Character Galaxy)
        self.meaning = meaning
        self.category = category
        self.references = 0  # Usage count

    def to_dict(self) -> Dict:
        return {
            "word_id": self.word_id,
            "characters": self.characters,  # List of char IDs (symlink pattern!)
            "meaning": self.meaning,
            "category": self.category,
            "references": self.references
        }

    @staticmethod
    def from_dict(data: Dict) -> 'SemanticWord':
        word = SemanticWord(
            word_id=data["word_id"],
            characters=data["characters"],
            meaning=data["meaning"],
            category=data["category"]
        )
        word.references = data.get("references", 0)
        return word


class SemanticVocabulary:
    """
    Vocabulary of semantic words (procedural composition).

    Stores words ONCE, referenced many times (symlink pattern).
    """

    def __init__(self):
        self.words: Dict[str, SemanticWord] = {}
        self._bootstrap_semantic_primitives()

    def _bootstrap_semantic_primitives(self):
        """Initialize with common ARC semantic concepts."""

        # Transformation types
        self._add_word("rotation_or_reflection", "Rotation or reflection transformation", "transformation")
        self._add_word("color_transformation", "Color mapping transformation", "transformation")
        self._add_word("pattern_repetition", "Pattern repetition transformation", "transformation")
        self._add_word("geometric_transformation", "Generic geometric transformation", "transformation")
        self._add_word("spatial_rearrangement", "Spatial rearrangement of elements", "transformation")

        # Structural conditions
        self._add_word("asymmetric_input", "Input has no symmetry", "condition")
        self._add_word("symmetric_input", "Input has symmetry", "condition")
        self._add_word("sparse_grid", "Grid with many empty cells", "condition")
        self._add_word("dense_grid", "Grid with few empty cells", "condition")
        self._add_word("multiple_objects", "Input has multiple objects", "condition")
        self._add_word("single_object", "Input has single object", "condition")

        # Pattern conditions
        self._add_word("rotation_task", "Task involves rotation", "pattern")
        self._add_word("reflection_task", "Task involves reflection", "pattern")
        self._add_word("color_change_task", "Task involves color changes", "pattern")
        self._add_word("repetition_task", "Task involves repetition", "pattern")
        self._add_word("border_task", "Task involves borders", "pattern")

        # Visual properties
        self._add_word("has_border", "Grid has border", "property")
        self._add_word("has_repetition", "Grid has repeating pattern", "property")
        self._add_word("connected_components", "Grid has connected components", "property")
        self._add_word("multiple_colors", "Grid has multiple colors", "property")

    def _add_word(self, word_id: str, meaning: str, category: str):
        """Add word as character sequence (procedural composition)."""
        # Convert word_id to character IDs (ASCII for now, could use procedural font IDs)
        characters = [ord(c) for c in word_id]

        word = SemanticWord(
            word_id=word_id,
            characters=characters,
            meaning=meaning,
            category=category
        )
        self.words[word_id] = word

    def ref(self, word_id: str) -> Optional[str]:
        """
        Create reference to word (symlink pattern).
        Returns word_id if exists, increments reference count.
        """
        if word_id in self.words:
            self.words[word_id].references += 1
            return word_id
        return None

    def resolve(self, word_id: str) -> Optional[SemanticWord]:
        """Resolve reference to actual word (follow symlink)."""
        return self.words.get(word_id)

    def get_or_create(self, word_id: str, meaning: str = "", category: str = "unknown") -> str:
        """Get existing word or create new one (deduplication)."""
        if word_id not in self.words:
            self._add_word(word_id, meaning or f"Semantic concept: {word_id}", category)
        return self.ref(word_id)

    def to_dict(self) -> Dict:
        return {
            "words": {word_id: word.to_dict() for word_id, word in self.words.items()},
            "total_words": len(self.words),
            "total_references": sum(word.references for word in self.words.values())
        }

    @staticmethod
    def from_dict(data: Dict) -> 'SemanticVocabulary':
        vocab = SemanticVocabulary()
        vocab.words = {
            word_id: SemanticWord.from_dict(word_data)
            for word_id, word_data in data.get("words", {}).items()
        }
        return vocab


class SemanticContext:
    """
    Semantic context using procedural word references.

    CORRECT: References words (character sequences), doesn't duplicate strings!
    """

    def __init__(self):
        self.contexts: List[Dict] = []
        self.vocabulary = SemanticVocabulary()

    def record_context(
        self,
        program: str,
        input_grid: np.ndarray,
        output_grid: Optional[np.ndarray],
        task_id: str,
        score: float
    ) -> Dict:
        """
        Record semantic context with word references (procedural!).
        """
        input_sig = SemanticSignature.extract(input_grid)
        output_sig = SemanticSignature.extract(output_grid) if output_grid is not None else {}

        # Infer transformation type
        transformation_type = SemanticSignature.compute_transformation_type(input_sig, output_sig)

        # Infer usage conditions
        when_to_use = self._infer_usage_conditions(input_sig, transformation_type)

        # Create context with WORD REFERENCES (not duplicate strings!)
        context = {
            "program": program,
            "task_id": task_id,
            "score": score,

            # PROCEDURAL: References to vocabulary words (symlink pattern!)
            "transformation_type_ref": self.vocabulary.ref(transformation_type),
            "when_to_use_refs": [self.vocabulary.ref(cond) for cond in when_to_use],

            # Signatures for matching
            "input_signature": input_sig,
            "output_signature": output_sig,
        }

        self.contexts.append(context)
        return context

    def _infer_usage_conditions(self, input_sig: Dict, transformation_type: str) -> List[str]:
        """Infer when this program should be used."""
        conditions = []

        # Structural conditions
        if input_sig.get("sparsity", 0.5) > 0.7:
            conditions.append(self.vocabulary.get_or_create("sparse_grid", category="condition"))

        has_symmetry = any([
            input_sig.get("symmetry_vertical"),
            input_sig.get("symmetry_horizontal"),
            input_sig.get("symmetry_diagonal")
        ])

        if has_symmetry:
            conditions.append(self.vocabulary.get_or_create("symmetric_input", category="condition"))
        else:
            conditions.append(self.vocabulary.get_or_create("asymmetric_input", category="condition"))

        # Pattern conditions based on transformation type
        if "rotation" in transformation_type:
            conditions.append(self.vocabulary.get_or_create("rotation_task", category="pattern"))
        if "color" in transformation_type:
            conditions.append(self.vocabulary.get_or_create("color_change_task", category="pattern"))
        if "repetition" in transformation_type:
            conditions.append(self.vocabulary.get_or_create("repetition_task", category="pattern"))

        # Visual properties
        if input_sig.get("has_border"):
            conditions.append(self.vocabulary.get_or_create("has_border", category="property"))
        if input_sig.get("has_repetition"):
            conditions.append(self.vocabulary.get_or_create("has_repetition", category="property"))

        return conditions

    def find_matching_contexts(
        self,
        query_grid: np.ndarray,
        top_k: int = 5,
        similarity_threshold: float = 0.5,
        use_fuzzy_matching: bool = True
    ) -> List[Dict]:
        """Find matching contexts with multi-tier similarity."""
        query_sig = SemanticSignature.extract(query_grid)

        matches = []
        for ctx in self.contexts:
            input_sig = ctx["input_signature"]

            # Multi-component similarity
            structural_sim = self._structural_similarity(query_sig, input_sig)
            color_sim = self._color_similarity(query_sig, input_sig)
            pattern_sim = self._pattern_similarity(query_sig, input_sig)

            # Weighted combination
            similarity = (
                0.4 * structural_sim +
                0.3 * pattern_sim +
                0.3 * color_sim
            )

            # Fuzzy matching
            if use_fuzzy_matching and similarity < similarity_threshold:
                if self._fuzzy_match(query_sig, input_sig):
                    similarity = max(similarity, 0.45)

            if similarity >= similarity_threshold:
                # Resolve word references to get readable context
                transformation_type = self.vocabulary.resolve(ctx["transformation_type_ref"])
                when_to_use_words = [
                    self.vocabulary.resolve(ref) for ref in ctx.get("when_to_use_refs", [])
                ]

                matches.append({
                    "program": ctx["program"],
                    "score": similarity,
                    "transformation_type": transformation_type.word_id if transformation_type else "unknown",
                    "when_to_use": [w.word_id for w in when_to_use_words if w],
                    "match_components": {
                        "structural": structural_sim,
                        "color": color_sim,
                        "pattern": pattern_sim
                    }
                })

        # Sort by score
        matches.sort(key=lambda x: x["score"], reverse=True)
        return matches[:top_k]

    def _structural_similarity(self, sig1: Dict, sig2: Dict) -> float:
        """Compare structural features."""
        score = 0.0

        # Symmetry matching
        for sym_type in ["symmetry_vertical", "symmetry_horizontal", "symmetry_diagonal"]:
            if sig1.get(sym_type) == sig2.get(sym_type):
                score += 0.33

        # Sparsity similarity
        sparsity_diff = abs(sig1.get("sparsity", 0.5) - sig2.get("sparsity", 0.5))
        if sparsity_diff < 0.2:
            score += 0.2

        return min(score, 1.0)

    def _color_similarity(self, sig1: Dict, sig2: Dict) -> float:
        """Compare color features."""
        score = 0.0

        # Number of colors
        if sig1.get("num_colors") == sig2.get("num_colors"):
            score += 0.5

        # Color distribution (Jaccard)
        colors1 = set(sig1.get("color_distribution", {}).keys())
        colors2 = set(sig2.get("color_distribution", {}).keys())

        if colors1 and colors2:
            jaccard = len(colors1 & colors2) / len(colors1 | colors2)
            score += 0.5 * jaccard

        return score

    def _pattern_similarity(self, sig1: Dict, sig2: Dict) -> float:
        """Compare pattern features."""
        score = 0.0

        # Connected components
        comp_diff = abs(sig1.get("connected_components", 0) - sig2.get("connected_components", 0))
        if comp_diff <= 2:
            score += 0.4

        # Border presence
        if sig1.get("has_border") == sig2.get("has_border"):
            score += 0.3

        # Repetition pattern
        if sig1.get("has_repetition") == sig2.get("has_repetition"):
            score += 0.3

        return min(score, 1.0)

    def _fuzzy_match(self, sig1: Dict, sig2: Dict) -> bool:
        """Fuzzy matching for edge cases."""
        # Same transformation type
        if sig1.get("transformation_type") == sig2.get("transformation_type"):
            return True

        # Similar dimensions
        dims1 = sig1.get("dimensions", (0, 0))
        dims2 = sig2.get("dimensions", (0, 0))

        if dims1[0] > 0 and dims2[0] > 0:
            ratio = min(dims1[0] / dims2[0], dims2[0] / dims1[0])
            if ratio > 0.5:
                return True

        return False

    def save(self, path: Path) -> None:
        """Save semantic context with vocabulary."""
        state = {
            "contexts": self.contexts,
            "vocabulary": self.vocabulary.to_dict(),
            "total_contexts": len(self.contexts)
        }

        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, 'w', encoding='utf-8') as f:
            json.dump(state, f, indent=2)

        # Print storage savings
        if self.vocabulary.words:
            total_refs = sum(w.references for w in self.vocabulary.words.values())
            total_words = len(self.vocabulary.words)
            if total_refs > 0:
                savings = (1 - total_words / total_refs) * 100
                print(f"[SemanticContext] Storage savings: {savings:.1f}% "
                      f"({total_words} words, {total_refs} references)")

    def load(self, path: Path) -> None:
        """Load semantic context with vocabulary."""
        if not path.exists():
            print(f"[SemanticContext] No checkpoint at {path}, starting fresh")
            return

        with open(path, 'r', encoding='utf-8') as f:
            state = json.load(f)

        self.contexts = state.get("contexts", [])
        if "vocabulary" in state:
            self.vocabulary = SemanticVocabulary.from_dict(state["vocabulary"])

        total_refs = sum(w.references for w in self.vocabulary.words.values())
        print(f"[SemanticContext] Loaded {len(self.contexts)} contexts, "
              f"{len(self.vocabulary.words)} words, {total_refs} references")
```

def main():
    # ... existing setup ...

    # DEEP TRAINING: Fewer tasks, MORE epochs, FREQUENT saves

    for cycle in range(args.cycles):
        print(f"\nCycle {cycle+1}/{args.cycles}...")

        for epoch in range(args.epochs):
            print(f"  Epoch {epoch+1}/{args.epochs}...")

            # Run epoch
            results = run_epoch(pipeline, tasks, args)

            # CRITICAL: Save after EVERY epoch (not just cycles!)
            print(f"  [SAVE] Saving checkpoints after epoch {epoch+1}...")
            pipeline.save_all()

            # Show progress
            correct = sum(1 for r in results if r["success"])
            print(f"  Epoch {epoch+1}: {correct}/{len(tasks)} correct ({correct/len(tasks)*100:.1f}%)")

            # Show library growth
            if hasattr(pipeline.shadow, 'deduplicator'):
                unique = len(pipeline.shadow.deduplicator.canonical_programs)
                refs = sum(len(u) for u in pipeline.shadow.deduplicator.usage_metadata.values())
                print(f"  Library: {unique} unique programs, {refs} total uses")

        # Prune after each cycle (not each epoch)
        if hasattr(pipeline.shadow, 'prune_low_quality'):
            print(f"\n[PRUNE] Cycle {cycle+1} complete, pruning low quality...")
            stats = pipeline.shadow.prune_low_quality()
            print(f"  Removed {stats.get('removed_programs', 0)} programs")
            print(f"  Remaining: {stats.get('unique_remaining', 0)} unique programs")

def run_epoch(pipeline, tasks, args):
    """Run one epoch over all tasks."""
    results = []

    for task_idx, task in enumerate(tasks):
        print(f"    Task {task_idx+1}/{len(tasks)}: {task['task_id'][:8]}...")

        # Process each test input
        for test_idx, test_input in enumerate(task['test']):
            result = pipeline.process_task(
                task_id=task['task_id'],
                test_input=test_input['input'],
                train_examples=task['train'],
                expected_output=test_input.get('output'),
                top_k=args.top_k
            )

            results.append({
                "task_id": task['task_id'],
                "test_idx": test_idx,
                "success": result.success if hasattr(result, 'success') else False
            })

    return results
```

**Key changes**:
1. Save after EVERY epoch (incremental accumulation)
2. Print progress after each epoch (visibility)
3. Show library growth (monitor accumulation)
4. Prune only after cycles (less aggressive)
