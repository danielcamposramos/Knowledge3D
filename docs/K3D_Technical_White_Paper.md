## 4.0 The Cranium Core: A GPU-Sovereign, Unified Multimodal Mind

The **Cranium Core** is the AI's central processing engine, governed by an uncompromising technical mandate: **GPU Sovereignty**. All critical-path operations—reasoning, spatial queries, and navigation—are executed exclusively in low-level, high-performance PTX (Parallel Thread Execution), with zero CPU fallbacks. This strict GPU-native approach is essential to meeting our stringent **<100ms latency target**. The credibility of this claim is not theoretical; it is grounded in production-grade PTX kernels, as demonstrated by this representative snippet from our frustum culling implementation:

```ptx
.entry warp_frustum_cull_simd(
    .param .u64 node_positions,
    .param .u32 node_count,
    .param .u64 visible_mask_out
) {
    .reg .u32 lane, idx, mask;
    .reg .f32 x, y, z, w, px, py, pz;
    .reg .pred inside;

    mov.u32 lane, %laneid;
    mov.u32 idx, lane;
    setp.ge.u32 p0, idx, node_count;
    @p0 bra warp_exit;

    // Load pos (coalesced) & project with view_proj matrix in const mem
    ld.global.f32 x, [node_positions + idx*12 + 0];
    ld.global.f32 y, [node_positions + idx*12 + 4];
    ld.global.f32 z, [node_positions + idx*12 + 8];

    // ... matrix multiplication logic ...

    // NDC bounds test (simplified for brevity)
    setp.ge.f32 p_x, px, -w;
    setp.le.f32 p_y, py, w;
    and.pred inside, p_x, p_y;

    // Vote result into a single 32-bit mask
    selp.u32 mask, 1, 0, inside;
    shl.b32 mask, mask, lane;

    // Warp-level OR reduction (no atomics, no bar.sync)
    shfl.sync.bfly.b32 %r_tmp, mask, 1, 0x1F; or.b32 mask, mask, %r_tmp;
    // ... further shfl reductions ...

    // Lane 0 writes final mask to global memory
    setp.eq.u32 p_lane0, lane, 0;
    @p_lane0 st.global.u32 [visible_mask_out], mask;

warp_exit:
    ret;
}
```
