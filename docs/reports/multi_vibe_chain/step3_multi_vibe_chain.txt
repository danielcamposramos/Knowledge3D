### 1. GPU Clustering: Affinity Propagation (AP) over K-Means
Why AP? K-means is centroid-forced (Euclidean bias on high-dim embeddings), but semantic domains crave exemplar-based (nodes "vote" for reps via cosine sim). AP's message-passing is warp-coop heaven: Parallel responsibility/availability updates, converges in ~10-20 iters. GPU-parallel: Each warp handles a node exemplar pair, shfl-reduce for max sims. Targets ~1k nodes/domain (28 domains for 28k nodes), balanced via damping.

**Algorithm Pseudocode (PTX-Host Hybrid):**
```python
# knowledge3d/spatial/domain_splitter.py
import cupy as cp
from cupyx.scipy.sparse import csr_matrix

class SemanticDomainSplitter:
    def __init__(self, sim_threshold=0.85, damping=0.9):
        self.sim_threshold = sim_threshold  # Cosine >0.85 = strong affinity
        self.damping = damping  # AP stability param

    def split_domains(self, embeddings_gpu, positions_gpu, edges_gpu, kb_limit=48):
        """
        GPU-AP clustering on embeddings → domains.
        Returns: domain_ids (N,), bridges (M x 2), domains list.
        """
        n_nodes = embeddings_gpu.shape[0]
        
        # Step 1: Bootstrap spatial hints via Morton levels (your octree insight)
        morton_levels = self._assign_morton_levels(positions_gpu)  # 0-10 levels → initial affinity boost
        
        # Step 2: Build similarity matrix (sparse, cosine on embeddings)
        # Use cp.dot(embeddings, embeddings.T) but sparse-ify >threshold
        sim_matrix = self._compute_sparse_cosine(embeddings_gpu)  # CSR: ~10% density for sem-close nodes
        sim_matrix = sim_matrix.multiply(self.sim_threshold)  # Binary mask for strong ties
        
        # Boost intra-level sims (spatial-semantic hybrid)
        level_mask = self._level_to_mask(morton_levels)  # Sparse diag for level affinity
        sim_matrix += level_mask  # Additive, keeps sparsity
        
        # Step 3: GPU-AP message passing (10 iters max)
        domain_ids = self._affinity_propagation_gpu(sim_matrix, n_iters=20)
        
        # Step 4: Validate balance (<kb_limit per domain)
        domain_sizes = cp.bincount(domain_ids)
        over_limit = domain_sizes > (kb_limit * 1024 / 4)  # Rough u32 edges est.
        if cp.any(over_limit):
            # Recursive split on large domains (rare, <5%)
            domain_ids = self._refine_large_domains(domain_ids, sim_matrix, over_limit)
        
        # Step 5: Extract bridges (cross-domain edges)
        src_dom = domain_ids[edges_gpu[:, 0]]
        dst_dom = domain_ids[edges_gpu[:, 1]]
        bridge_mask = src_dom != dst_dom
        bridges = edges_gpu[bridge_mask]
        
        # Build per-domain kernels
        domains = []
        for d in cp.unique(domain_ids):
            dom_mask = domain_ids == d
            dom_edges = edges_gpu[dom_mask]  # Intra only
            domains.append(self._build_led_kernel(dom_edges))  # <48KB CSR
        
        return domain_ids, bridges, domains

    def _affinity_propagation_gpu(self, sim_csr, n_iters=20):
        """Warp-coop AP: Parallel max-responsibility/availability."""
        n = sim_csr.shape[0]
        r = cp.zeros(n, dtype=cp.float32)  # Responsibilities
        a = cp.zeros(n, dtype=cp.float32)  # Availabilities
        
        for iter in range(n_iters):
            # Parallel: For each node i, compute r_i,k = sim_i,k - max_{k'!=k} sim_i,k' (warp max excl self)
            for i in range(0, n, 32):  # Warp stride
                warp_r = self._warp_max_excl(sim_csr[i:i+32].data, sim_csr.indices[i:i+32])
                r[i:i+32] = sim_csr.data[i:i+32] - warp_r  # Approx; full via cub::BlockReduce
            
            # Update a: Sum r over exemplars, warp-sum
            a = self._warp_sum_positive(r) * self.damping + a * (1 - self.damping)
            
            # Converge check: max |delta| < epsilon
            if self._converged(r, a):
                break
        
        # Exemplars as domains (argmax a_i,i)
        domain_centers = cp.argmax(cp.diag(a), axis=0)
        domain_ids = cp.full(n, -1)
        for center in domain_centers:
            mask = cp.isclose(a[:, center], a.max(axis=1, keepdims=True), atol=1e-5)
            domain_ids[mask] = center  # Assign to nearest exemplar
        
        return domain_ids

    def _assign_morton_levels(self, positions_gpu):
        """Hybrid: Morton-derived levels as affinity prior (0=root, 10=leaf)."""
        # From your 10-bit: level = popcount(morton_code) // 3 (x/y/z bits)
        morton_codes = self.compute_morton_codes(positions_gpu)  # Your PTX call
        levels = cp.popcount(morton_codes) // 3  # 0-10
        return levels

    # Helpers: _compute_sparse_cosine (cp.dot sparse), _warp_max_excl (shfl excl), etc.
```

**PTX Stub for Warp-Max-Excl (Core of AP Parallelism):**
```ptx
// knowledge3d/cuda/warp_max_excl.ptx (<0.5KB)
.entry warp_max_excl(
    .param .u64 sim_values,  // Per-warp sims (32 f32)
    .param .u64 indices,     // Excl self idx
    .param .u64 max_out      // Output max excl
) {
    .reg .f32 sim, max_sim;
    .reg .u32 lane;
    mov.u32 lane, %laneid;
    ld.global.f32 sim, [sim_values + lane*4];
    
    // Warp reduce max, but excl self (shfl to neighbor, max across)
    shfl.sync.b32 max_sim, sim, lane ^ 1, 32;  // Swap with neighbor for excl proxy
    // Full: Use cub::WarpReduceMax on masked (all but self)
    vote.ballot.sync $p, lane != excl_idx;  // Mask
    reduce.max.sync.f32 max_sim, sim, $p;   // Conditional reduce
    
    @eq.laneid0 st.global.f32 [max_out + %warpid*4], max_sim;
}
```

### 2. Bridge Edge Detection Refinement: Minimal Cut via Sem-Spatial Threshold
Bridges = cross-domain edges with high sem-sim (>0.85) *and* spatial boundary cross (Morton level diff >2). This prunes 80% false bridges (pure spatial jumps without sem tie). GPU-parallel: Per-edge warp-check (src_morton ^ dst_morton popcount > threshold for boundary).

**Refinement Pseudocode:**
```python
def _find_cross_domain_edges(self, edges_gpu, domain_ids, morton_codes_gpu):
    src_dom = domain_ids[edges_gpu[:, 0]]
    dst_dom = domain_ids[edges_gpu[:, 1]]
    cross_dom_mask = src_dom != dst_dom
    
    # Sem filter: cosine >0.85 (precomputed or quick dot)
    sem_sim = self._batch_cosine(embeddings_gpu[edges_gpu[:, 0]], embeddings_gpu[edges_gpu[:, 1]])
    sem_mask = sem_sim > 0.85
    
    # Spatial boundary: popcount(src_morton ^ dst_morton) > 6 (level diff proxy)
    xor_morton = cp.bitwise_xor(morton_codes_gpu[edges_gpu[:, 0]], morton_codes_gpu[edges_gpu[:, 1]])
    boundary_mask = cp.popcount(xor_morton) > 6  # ~level 2+ diff
    
    bridge_mask = cross_dom_mask & sem_mask & boundary_mask
    return edges_gpu[bridge_mask]  # ~5-10% of edges, <1KB constant mem
```

```ptx
// knowledge3d/cuda/enhanced_affinity_propagation.ptx

// Optimized AP with warp-cooperative message passing
.entry enhanced_affinity_propagation(
    .param .u64 similarity_matrix,    // Sparse CSR format
    .param .u64 responsibilities,     // R matrix output
    .param .u64 availabilities,       // A matrix output
    .param .u64 domain_ids,          // Final domain assignments
    .param .u32 n_nodes,
    .param .u32 max_iterations,
    .param .f32 convergence_threshold,
    .param .f32 damping_factor
)
{
    .reg .u32 warp_id, lane_id;
    .reg .u32 iteration, converged;
    .reg .f32 damp, one_minus_damp;
    .reg .pred should_continue;
    
    mov.u32 warp_id, %warpid;
    mov.u32 lane_id, %laneid;
    ld.param.f32 damp, [damping_factor];
    sub.f32 one_minus_damp, 1.0, damp;
    
    // Initialize matrices
    call initialize_matrices, (responsibilities, availabilities, n_nodes);
    
    mov.u32 iteration, 0;
    
ap_loop:
    // Phase 1: Update responsibilities
    call update_responsibilities_warp, (similarity_matrix, responsibilities, 
                                       availabilities, n_nodes, damp);
    
    // Phase 2: Update availabilities  
    call update_availabilities_warp, (responsibilities, availabilities, 
                                     n_nodes, damp, one_minus_damp);
    
    // Phase 3: Check convergence
    call check_convergence_warp, (responsibilities, availabilities, 
                                  convergence_threshold, converged);
    
    // Phase 4: Extract exemplars if converged
    @converged call extract_exemplars_warp, (responsibilities, availabilities, 
                                           domain_ids, n_nodes);
    
    add.u32 iteration, iteration, 1;
    setp.lt.u32 should_continue, iteration, max_iterations;
    @should_continue bra ap_loop;
    
    ret;
}

// Warp-cooperative responsibility update
.func update_responsibilities_warp(
    .param .u64 sim_matrix,
    .param .u64 responsibilities,
    .param .u64 availabilities,
    .param .u32 n_nodes,
    .param .f32 damping
)
{
    .reg .u32 warp_id, lane_id;
    .reg .u32 node_idx, max_sim_idx;
    .reg .f32 max_sim, current_sim, new_resp;
    .reg .f32 avail_sum;
    
    mov.u32 warp_id, %warpid;
    mov.u32 lane_id, %laneid;
    
    // Each warp processes a block of nodes
    mov.u32 node_idx, warp_id * 32 + lane_id;
    
    // Load similarity values for this node
    call load_node_similarities, (sim_matrix, node_idx);
    
    // Find maximum similarity (excluding self) via warp reduction
    call warp_max_similarity_excl_self, (max_sim, max_sim_idx);
    
    // Update responsibility: r(i,k) = s(i,k) - max_{k'≠k} a(i,k') + s(i,k')
    ld.global.f32 current_sim, [sim_matrix + node_idx * 4];
    sub.f32 new_resp, current_sim, max_sim;
    
    // Apply damping
    mul.f32 new_resp, new_resp, damping;
    
    // Store updated responsibility
    st.global.f32 [responsibilities + node_idx * 4], new_resp;
    
    ret;
}
```

```ptx
// knowledge3d/cuda/optimized_bridge_storage.ptx

// Constant memory layout for 1-cycle bridge access
.const .align 8 .b8 bridge_constant_table[4096];  // 4KB for bridges

.struct BridgeLookup {
    .u32 domain_offsets[32];    // Offset per domain (max 32 domains)
    .u32 bridge_counts[32];     // Bridge count per domain
    .f32 bridge_weights[1024];  // Bridge weights (32 * 32 max)
    .u32 bridge_targets[1024];  // Bridge target nodes
};  // Total: 4KB + 4KB + 4KB + 4KB = 16KB

// Ultra-fast bridge lookup
.entry fast_bridge_lookup(
    .param .u32 source_domain,
    .param .u32 target_domain,
    .param .u64 bridge_weight_out,
    .param .u64 bridge_target_out
)
{
    .reg .u32 domain_offset, bridge_count;
    .reg .u32 bridge_idx, target_node;
    .reg .f32 bridge_weight;
    .reg .pred bridge_found;
    
    // Load domain offset from constant memory (1 cycle)
    ld.const.u32 domain_offset, [bridge_constant_table + source_domain * 4];
    ld.const.u32 bridge_count, [bridge_constant_table + source_domain * 4 + 128];
    
    // Search for bridge to target domain
    mov.u32 bridge_idx, 0;
    
bridge_search_loop:
    setp.ge.u32 search_done, bridge_idx, bridge_count;
    @search_done bra bridge_not_found;
    
    // Load bridge entry
    ld.const.u32 target_node, [bridge_constant_table + domain_offset + bridge_idx * 8];
    ld.const.f32 bridge_weight, [bridge_constant_table + domain_offset + bridge_idx * 8 + 4];
    
    // Check if this is the target domain
    setp.eq.u32 bridge_found, target_node, target_domain;
    @bridge_found bra bridge_found;
    
    add.u32 bridge_idx, bridge_idx, 1;
    bra bridge_search_loop;
    
bridge_found:
    // Store results (1 cycle each)
    st.global.u32 [bridge_target_out], target_node;
    st.global.f32 [bridge_weight_out], bridge_weight;
    bra lookup_complete;
    
bridge_not_found:
    // No bridge found - store sentinel values
    st.global.u32 [bridge_target_out], 0xFFFFFFFF;
    st.global.f32 [bridge_weight_out], 0.0;
    
lookup_complete:
    ret;
}
```

```ptx
// Stitch intra-domain paths with bridge traversals
.entry stitch_cross_domain_path(
    .param .u64 source_domain_path,
    .param .u64 target_domain_path,
    .param .u64 bridge_sequence,
    .param .u64 stitched_path_out,
    .param .u32 max_path_length
)
{

    .reg .u32 total_len, path_idx;
    .reg .pred path_valid;
```

```python
def _compute_sparse_cosine(self, embeddings_gpu):
    """Sparsity-pruned cosine: Bitmask low-sims, warp-dot survivors."""
    n = embeddings_gpu.shape[0]
    dim = embeddings_gpu.shape[1]
    
    # Step 1: Approx dot via warp-sum (norms precomputed)
    norms = cp.linalg.norm(embeddings_gpu, axis=1, keepdims=True)
    approx_dots = cp.dot(embeddings_gpu, embeddings_gpu.T) / (norms @ norms.T)  # Full dense temp—avoid!
    
    # Prune: Warp-vote mask for >0.1 (your threshold vibe)
    mask = cp.abs(approx_dots) > 0.1  # Binary CSR from dense (temp ok, 28k=0.8MB)
    approx_dots = approx_dots * mask  # Zero low tails
    
    # Step 2: Fused warp-dot on masked pairs only
    sim_data, sim_indices = [], []
    for i in range(0, n, 32):  # Warp blocks
        block_mask = mask[i:i+32]
        survivor_cols = cp.nonzero(block_mask)[1]  # ~10% per row
        for j in survivor_cols:
            cos_sim = self._warp_cosine(embeddings_gpu[i], embeddings_gpu[j])  # PTX call
            if cos_sim > self.sim_threshold:
                sim_data.append(cos_sim)
                sim_indices.append(j)
    
    return csr_matrix((sim_data, (range(len(sim_data)), sim_indices)), shape=(n, n))
```

```ptx
// knowledge3d/cuda/warp_cosine.ptx
.entry warp_cosine(
    .param .u64 vec_a, .param .u64 vec_b,  // 128-dim f32 per vec
    .param .u32 dim,
    .param .u64 cos_out  // Output f32 cosine
) {
    .reg .f32 dot_prod, norm_a, norm_b, cos_sim;
    .reg .u32 lane, i;
    mov.u32 lane, %laneid;
    
    // Warp-fused dot: shfl vec_b to lanes, FMA sum
    mov.f32 dot_prod, 0.0;
    mov.u32 i, 0;
 dot_loop:
    setp.ge.u32 p, i, dim;
    @p bra dot_done;
    
    ld.global.f32 $f0, [vec_a + i*4 + lane*dim*4];  // Stride for warp
    shfl.sync.b32 $f1, $f0, lane ^ 1, 32;  // Broadcast b equiv
    ld.global.f32 $f2, [vec_b + i*4];
    fma.rn.f32 dot_prod, $f0, $f2, dot_prod;
    add.u32 i, i, 1;
    bra dot_loop;
    
 dot_done:
    // Warp-reduce dot (cub::WarpReduceSum proxy)
    reduce.add.sync.f32 dot_prod, dot_prod;
    
    // Norms (precomp or inline sqrt-sum-sq)
    call warp_norm, (vec_a, dim, norm_a);
    call warp_norm, (vec_b, dim, norm_b);
    div.approx.f32 cos_sim, dot_prod, (norm_a * norm_b);
    
    @eq.laneid0 st.global.f32 [cos_out], cos_sim;
}
```
